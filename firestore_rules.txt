rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }

    // Lobbies collection
    match /lobbies/{lobbyCode} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.resource.data.hostUid == request.auth.uid;
      allow update, delete: if isAuthenticated();

      // Players subcollection
      match /players/{playerUid} {
        allow read: if isAuthenticated();
        allow create, update, delete: if isAuthenticated();
      }

      // Rounds subcollection
      match /rounds/{roundIndex} {
        allow read: if isAuthenticated();
        allow create, update, delete: if isAuthenticated();

        // Guesses subcollection
        match /guesses/{playerUid} {
          allow read: if isAuthenticated();
          allow create, update, delete: if isAuthenticated();
        }
      }
    }
  }
}
